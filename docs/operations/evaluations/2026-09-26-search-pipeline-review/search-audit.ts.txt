import postgres from 'postgres';
import {writeFileSync} from 'node:fs';
import {profileEvidence,profileHash} from '@/lib/domain/products/search-profile';
import {pendingProfiles,pendingVerifications} from '@/lib/domain/products/search-profiles';
import {pendingIntroChecks} from '@/lib/domain/products/intro-checks';
const c=postgres(process.env.DATABASE_URL!,{max:1,prepare:false,onnotice:()=>{},debug:(_c,q)=>console.error("SQL",String(q).slice(0,110))});(globalThis as any).pgClient=c;
async function main(){
const data=await c.begin('read only',async s=>{
await s`set local jit=off`;await s`set local statement_timeout='20s'`;
const publicWhere="p.status in ('seeded','verified')";
const coverage=await s.unsafe(`select count(*)::int as total,count(*) filter(where p.source='crawler')::int as crawler,count(*) filter(where nullif(btrim(search_topics),'') is not null)::int as topics,count(*) filter(where nullif(btrim(search_page_text),'') is not null)::int as page,count(*) filter(where nullif(btrim(search_readme),'') is not null)::int as readme,count(*) filter(where nullif(btrim(search_keywords),'') is not null)::int as keywords,count(*) filter(where nullif(btrim(search_category),'') is not null)::int as category,count(*) filter(where search_vector is null or search_vector=''::tsvector)::int as empty_vector,count(*) filter(where length(coalesce(search_page_text,''))<100 and length(coalesce(search_readme,''))<200)::int as thin from products p where ${publicWhere}`);
const sources=await s.unsafe(`select p.source,count(*)::int as n,count(*) filter(where c.repo is not null)::int as candidate,count(*) filter(where d.repo is not null)::int as document,count(*) filter(where d.repo is null and p.source='crawler')::int as missing_doc from products p left join crawl_candidates c on c.published_slug=p.slug left join crawl_documents d on d.repo=c.repo where ${publicWhere} group by p.source`);
const sourceComparison=await s.unsafe(`with paired as(select p.*,d.repo_meta,d.page_meta,case when jsonb_typeof(d.repo_meta->'topics')='array' then(select string_agg(topic,' ') from jsonb_array_elements_text(d.repo_meta->'topics') t(topic)) else null end as topics_expected,left(d.page_meta->>'textSample',2000) as page_expected,nullif(left(d.page_meta->>'readmeSample',2000),'') as readme_expected from products p join crawl_candidates c on c.published_slug=p.slug join crawl_documents d on d.repo=c.repo where ${publicWhere}) select count(*)::int as paired,count(*) filter(where search_topics is distinct from topics_expected)::int as topics_diff,count(*) filter(where search_page_text is distinct from page_expected)::int as page_diff,count(*) filter(where search_readme is distinct from readme_expected)::int as readme_diff,count(*) filter(where nullif(btrim(search_page_text),'') is null and nullif(btrim(page_expected),'') is not null)::int as page_lost,count(*) filter(where nullif(btrim(search_readme),'') is null and nullif(btrim(readme_expected),'') is not null)::int as readme_lost,count(*) filter(where nullif(btrim(page_expected),'') is null)::int as source_page_empty,count(*) filter(where nullif(btrim(readme_expected),'') is null)::int as source_readme_empty,count(*) filter(where page_meta->>'readmeSampleVersion' is null)::int as no_readme_version from paired`);
const profiles=await s.unsafe(`select count(*)::int as products,count(*) filter(where sp.product_id is null)::int as missing_profile,count(*) filter(where sp.error_code is not null)::int as failed,count(*) filter(where sp.error_code is not null and sp.attempts>=5)::int as exhausted,count(*) filter(where sp.product_id is not null and sp.error_code is null and jsonb_array_length(sp.keywords_en)+jsonb_array_length(sp.keywords_ko)=0)::int as empty_success,count(*) filter(where sp.error_code is null and jsonb_array_length(sp.keywords_en)=0)::int as empty_en,count(*) filter(where sp.error_code is null and jsonb_array_length(sp.keywords_ko)=0)::int as empty_ko,count(*) filter(where sp.verified_at is not null)::int as verified,count(*) filter(where sp.verified_at is null and sp.error_code is null and jsonb_array_length(sp.keywords_en)+jsonb_array_length(sp.keywords_ko)>0)::int as unverified,count(*) filter(where sp.verified_at is null and sp.verify_attempts>=5)::int as verify_exhausted,count(*) filter(where sp.verified_at is null and sp.verify_error is not null)::int as verify_failed,count(*) filter(where sp.product_id is null and p.search_keywords is not null)::int as orphan_keywords,count(*) filter(where sp.updated_at>now()-interval '1 hour' and sp.error_code is null)::int as generated1h,count(*) filter(where sp.verified_at>now()-interval '1 hour')::int as verified1h from products p left join product_search_profiles sp on sp.product_id=p.id where ${publicWhere}`);
const keywordCopy=await s.unsafe(`with expected as(select p.id,p.slug,p.search_keywords,nullif((select string_agg(value,' · ' order by ord) from jsonb_array_elements_text(sp.keywords_en || sp.keywords_ko) with ordinality a(value,ord) where not(sp.removed_keywords ? value)),'') as wanted from products p join product_search_profiles sp on sp.product_id=p.id where ${publicWhere} and sp.error_code is null) select count(*)::int as checked,count(*) filter(where search_keywords is distinct from wanted)::int as mismatched,count(*) filter(where search_keywords is null and wanted is not null)::int as lost from expected`);
const profileErrors=await s.unsafe(`select sp.error_code,sp.verify_error,count(*)::int as n,max(sp.updated_at)::text as latest from products p join product_search_profiles sp on sp.product_id=p.id where ${publicWhere} and (sp.error_code is not null or sp.verify_error is not null) group by 1,2 order by n desc`);
const verificationModels=await s.unsafe(`select verify_model,count(*)::int as n,min(sp.verified_at)::text as first,max(sp.verified_at)::text as latest from products p join product_search_profiles sp on sp.product_id=p.id where ${publicWhere} and sp.verified_at is not null group by 1 order by n desc`);
const intros=await s.unsafe(`select p.tagline_source,count(*)::int as n,count(*) filter(where i.product_id is not null)::int as checked,count(*) filter(where i.outcome='replaced')::int as replaced,count(*) filter(where i.outcome='needs_editor')::int as needs_editor from products p left join product_intro_checks i on i.product_id=p.id where ${publicWhere} group by 1 order by n desc`);
const schema=await s`select column_name,data_type,is_generated,generation_expression from information_schema.columns where table_schema='public' and table_name='products' and column_name like 'search_%' order by column_name`;
const indexes=await s`select indexname,indexdef from pg_indexes where schemaname='public' and (tablename='products' and indexdef like '%search%' or tablename in ('product_search_profiles','product_intro_checks','search_queries','text_translations')) order by tablename,indexname`;
const logStats=await s`select count(*)::int as total,count(*) filter(where searched_at>now()-interval '24 hours')::int as recent24h,count(*) filter(where searched_at>now()-interval '24 hours' and results=0 and not filtered)::int as zero_unfiltered24h,count(*) filter(where searched_at>now()-interval '24 hours' and query ~ '[가-힣]' and keywords is null)::int as untranslated_korean24h,min(searched_at)::text as first,max(searched_at)::text as latest,count(*) filter(where normalized is null or duration_ms is null)::int as incomplete from search_queries`;
const translationStats=await s`select target_lang,status,count(*)::int as n,max(updated_at)::text as latest from text_translations group by 1,2`;
const activeInvalid=await s`with latest as(select distinct on(repository_key,scope) * from agent_repository_scans order by repository_key,scope,started_at desc,id desc), demand as(select lower(repo) repository_key from crawl_candidates where state in('new','approved','needs_review') union select lower(split_part(repo_url,'/',4)||'/'||split_part(repo_url,'/',5)) from products where status<>'banned' and repo_url ~ '^https://github.com/[^/]+/[^/#?]+/?$') select state,last_error_code,count(*)::int as n from latest join demand using(repository_key) group by 1,2 order by n desc`;
const invalidSamples=await s`select repository_key,cursor->'pendingCommits'->0->>'sha' as commit_sha,cursor->'pendingCommits'->0->'observations' is not null as has_commit_observations,request_count,last_error_code from agent_repository_scans where last_error_code='invalid' order by started_at desc limit 4`;
return {at:new Date().toISOString(),coverage,sources,sourceComparison,profiles,keywordCopy,profileErrors,verificationModels,intros,schema,indexes,logStats,translationStats,activeInvalid,invalidSamples};});
// Selection functions issue SELECT only; their exact queue predicates are reused for the audit.
const readyProfiles=await pendingProfiles(25000);const readyVerify=await pendingVerifications(25000);const readyIntro=await pendingIntroChecks(25000);
const hashRows=await c`select p.name,p.url,p.category,p.tagline,p.description,p.search_topics,p.search_page_text,p.search_readme,sp.source_hash,(select a.ai_reason from product_audit_items a where a.product_id=p.id and a.ai_reason is not null order by a.id desc limit 1) as reviewer_note from products p join product_search_profiles sp on sp.product_id=p.id where p.status in('seeded','verified') and sp.error_code is null`;
let hashMismatch=0;for(const r of hashRows){const e=profileEvidence({name:r.name,url:r.url,category:r.category,tagline:r.tagline,description:r.description,searchTopics:r.search_topics,searchPageText:r.search_page_text,searchReadme:r.search_readme},r.reviewer_note);if(profileHash(e)!==r.source_hash)hashMismatch++;}
const result={...data,queues:{profiles:readyProfiles.length,verify:readyVerify.length,intro:readyIntro.length},profileHash:{checked:hashRows.length,mismatch:hashMismatch}};
writeFileSync('/tmp/nmv-search-audit-20260926.json',JSON.stringify(result,null,2));console.log(JSON.stringify(result,null,2));}
main().catch(e=>{console.error(JSON.stringify({name:e.name,code:e.code,message:e.message,position:e.position}));process.exitCode=1}).finally(()=>c.end());
