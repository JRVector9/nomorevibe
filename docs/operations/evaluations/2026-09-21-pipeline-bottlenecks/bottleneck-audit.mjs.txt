import {parseEnv} from 'node:util';import{readFileSync,writeFileSync}from'node:fs';import postgres from'postgres';
const db=postgres(parseEnv(readFileSync('/tmp/nmv-installable-web.env','utf8')).DATABASE_URL,{max:1,prepare:false,connect_timeout:10});
const result={at:new Date().toISOString()};
async function q(name,fn){const at=performance.now();try{result[name]=await db.begin('read only',async s=>{await s`set local statement_timeout='20s'`;return fn(s);});}catch(e){result[name]={error:e.code??e.name};}result[name+'QueryMs']=Math.round(performance.now()-at);}
try{
await q('settings',s=>s`select values->>'enabled' as enabled,values->>'reviewMode' as review_mode,values->>'reviewConcurrency' as concurrency,values->'agentEvidence' as agent_evidence,values->'firstReview' as first_review,values->'secondReview' as second_review from crawl_settings`);
await q('candidates',s=>s`select state,reason,count(*)::int as n,round(extract(epoch from (now()-min(updated_at)))/3600,2) as oldest_hours from crawl_candidates group by state,reason order by n desc`);
await q('publicHealth',s=>s`select p.access_mode,count(*)::int as n,count(*) filter(where h.checked_at is null)::int as never_checked,count(*) filter(where h.checked_at<now()-interval '6 hours')::int as older_6h,percentile_cont(.5) within group(order by extract(epoch from now()-h.checked_at)/3600) as age_p50_hours,max(extract(epoch from now()-h.checked_at)/3600) as max_age_hours from products p left join product_health h using(slug) where p.status in ('seeded','verified') group by p.access_mode`);
await q('stars',s=>s`select count(*)::int as public_repo_products,count(*) filter(where stars_at is null)::int as never_checked,count(*) filter(where stars_at<now()-interval '24 hours')::int as older_24h,count(*) filter(where (stars_at is null or stars_at<now()-interval '24 hours') and (stars_checked_at is null or stars_checked_at<now()-interval '1 hour'))::int as due,percentile_cont(.5) within group(order by extract(epoch from now()-stars_at)/3600) as age_p50_hours,percentile_cont(.95) within group(order by extract(epoch from now()-stars_at)/3600) as age_p95_hours from products where status in ('seeded','verified') and repo_url is not null`);
await q('sources',s=>s`select e.kind,e.state,e.last_error_code,count(*)::int as n,count(*) filter(where e.next_attempt_at<=now())::int as due,count(*) filter(where e.last_success_at is null)::int as never_succeeded,percentile_cont(.5) within group(order by extract(epoch from now()-e.last_success_at)/3600) as success_age_p50_h from product_evidence_sources e join products p using(slug) where p.status<>'banned' group by 1,2,3 order by n desc`);
await q('agentScans',s=>s`with latest as(select distinct on(repository_key) * from agent_repository_scans where scope='' order by repository_key,started_at desc,id desc) select state,last_error_code,count(*)::int as n,count(*) filter(where next_attempt_at<=now())::int as due,percentile_cont(.5) within group(order by extract(epoch from now()-completed_at)/3600) as age_p50_hours from latest group by 1,2 order by n desc`);
await q('taglines',s=>s`select count(*)::int as held,count(*) filter(where t.repo is null)::int as untried,count(*) filter(where t.error_code is not null)::int as errors,count(*) filter(where t.tagline='' and t.error_code is null)::int as empty,count(*) filter(where t.written_by is null and (t.repo is null or(t.error_code is not null and t.attempts<5 and(t.retry_at is null or t.retry_at<=now()))or t.document_at is distinct from d.fetched_at))::int as eligible from crawl_candidates c join crawl_documents d using(repo) left join crawl_taglines t using(repo) where c.state='needs_review' and c.reason='no_description' and c.decided_by='auto'`);
await q('classification',s=>s`select count(*)::int as holds,count(*) filter(where cd.retry_at>now())::int as waiting,count(*) filter(where cd.retry_at<=now())::int as due from category_decisions cd join crawl_candidates c using(repo) where c.state='approved' and cd.category is null`);
await q('translations',s=>s`with src as(select coalesce(a.reason,a.outcome->>'reason') as body,coalesce(a.completed_at,a.started_at) as at from crawl_review_attempts a where coalesce(a.reason,a.outcome->>'reason') is not null union all select second_reason,coalesce(reviewed_at,created_at) from second_reviews where second_reason is not null),u as(select encode(sha256(convert_to(body,'UTF8')),'hex') as hash,max(at) as at from src where length(btrim(body))>0 and(body!~'[가-힣]' or length(regexp_replace(body,'[^가-힣]','','g'))<.3*greatest(1,length(regexp_replace(body,'[^A-Za-z가-힣]','','g'))))group by 1)select count(*)::int as total,count(*) filter(where t.status='done')::int as done,count(*) filter(where t.status='failed')::int as failed,count(*) filter(where t.source_hash is null)::int as untried,count(*) filter(where u.at>now()-interval '1 hour')::int as recent_source_hashes,extract(epoch from now()-min(u.at)filter(where t.source_hash is null))/3600 as oldest_untried_h from u left join text_translations t on t.source_hash=u.hash and t.target_lang='ko'`);
await q('secondQueue',s=>s`select status,count(*)::int as rows,count(distinct candidate_id)::int as candidates,extract(epoch from now()-min(created_at))/3600 as oldest_h from second_reviews where status in ('pending','failed','agreed','needs_human')group by status`);
await q('admin24h',s=>s`select action,count(*)::int as n from operations_audit where created_at>now()-interval '24 hours' group by action order by n desc`);
await q('frontier',s=>s`select state,count(*)::int as n from crawl_frontier group by state`);
await q('db',s=>s`select current_setting('max_connections')::int as max_connections,(select count(*)::int from pg_stat_activity)as all_connections,(select count(*)::int from pg_stat_activity where datname=current_database())as app_connections,(select count(*)::int from pg_stat_activity where datname=current_database() and wait_event_type='Lock')as lock_waits,(select max(extract(epoch from now()-query_start))from pg_stat_activity where datname=current_database()and state='active' and pid<>pg_backend_pid())as longest_active_seconds`);
await q('jobs',s=>s`select name,last_error is not null as has_error,requested_version-processed_version as pending_versions,round(extract(epoch from now()-last_success_at))::int as success_age_seconds from jobs order by name`);
writeFileSync('/tmp/nmv-bottleneck-audit.json',JSON.stringify(result,null,2));console.log(JSON.stringify(result,null,2));
}finally{await db.end({timeout:5});}
